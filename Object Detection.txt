import matplotlib.pyplot as plt
import numpy as np
from tensorflow.keras.applications.vgg16 import VGG16, decode_predictions, preprocess_input
from tensorflow.keras.preprocessing.image import img_to_array, load_img

model = VGG16()
image_files = ['download.jpg', 'download2.png', 'download3.jpg']

fig, axes = plt.subplots(3, 2, figsize=(12, 10))

for i, path in enumerate(image_files):
  orig = load_img(path)
  x = preprocess_input(
      np.expand_dims(img_to_array(load_img(path, target_size=(224, 224))), 0)
  )

   top5 = decode_predictions(model.predict(x), top=5)[0][::-1]
  labels = [p[1].replace('_', ' ').title() for p in top5]
  scores = [p[2] * 100 for p in top5]

   axes[i, 0].imshow(orig)
  axes[i, 0].set_title(f'{path}\nTop: {labels[-1]} ({scores[-1]:.1f}%)')
  axes[i, 0].axis('off')

  axes[i, 1].barh(labels, scores, color='steelblue')
  axes[i, 1].set(
      xlim=(0, 100), xlabel='Confidence (%)', title='Top 5 Predictions'
  )

plt.tight_layout()
plt.show()
